CREATE DATABASE IF NOT EXISTS student_management;
USE student_management;

-- =========================================
-- 1. STUDENTS TABLE
-- =========================================

CREATE TABLE students (
    student_id INT PRIMARY KEY AUTO_INCREMENT,
    first_name VARCHAR(50) NOT NULL,
    last_name VARCHAR(50) NOT NULL,
    email VARCHAR(100) UNIQUE,
    phone VARCHAR(15),
    gender VARCHAR(10),
    date_of_birth DATE,
    city VARCHAR(50)
);

-- =========================================
-- 2. COURSES TABLE
-- =========================================

CREATE TABLE courses (
    course_id INT PRIMARY KEY AUTO_INCREMENT,
    course_name VARCHAR(100) NOT NULL,
    department VARCHAR(100),
    duration_months INT,
    fees DECIMAL(10,2)
);

-- =========================================
-- 3. ENROLLMENTS TABLE
-- =========================================

CREATE TABLE enrollments (
    enrollment_id INT PRIMARY KEY AUTO_INCREMENT,
    student_id INT NOT NULL,
    course_id INT NOT NULL,
    enrollment_date DATE,
    marks DECIMAL(5,2),
    grade VARCHAR(5),
    status VARCHAR(20),

    FOREIGN KEY (student_id) REFERENCES students(student_id),
    FOREIGN KEY (course_id) REFERENCES courses(course_id)
);

-- 7. BASIC READ QUERIES
-- =========================================

SELECT * FROM students;

SELECT student_id, first_name, last_name, city
FROM students;

SELECT *
FROM students
WHERE city = 'Pune';

SELECT first_name, last_name, city
FROM students
WHERE gender = 'Female';

SELECT student_id, marks, grade
FROM enrollments
WHERE marks > 80;

-- =========================================
-- 8. ORDER BY
-- =========================================

SELECT student_id, marks, grade
FROM enrollments
ORDER BY marks DESC;

SELECT student_id, marks, grade
FROM enrollments
ORDER BY marks ASC;

-- =========================================
-- 9. JOIN REPORTS
-- =========================================

SELECT
    s.student_id,
    CONCAT(s.first_name, ' ', s.last_name) AS student_name,
    c.course_name,
    e.enrollment_date,
    e.marks,
    e.grade,
    e.status
FROM students s
INNER JOIN enrollments e
    ON s.student_id = e.student_id
INNER JOIN courses c
    ON e.course_id = c.course_id;

-- =========================================
-- 10. AGGREGATE FUNCTIONS
-- =========================================

SELECT COUNT(*) AS total_students
FROM students;

SELECT COUNT(*) AS total_enrollments
FROM enrollments;

SELECT ROUND(AVG(marks), 2) AS average_marks
FROM enrollments;

-- =========================================
-- 11. GROUP BY REPORTS
-- =========================================

SELECT
    c.course_name,
    COUNT(e.student_id) AS total_students
FROM courses c
LEFT JOIN enrollments e
    ON c.course_id = e.course_id
GROUP BY c.course_id, c.course_name
ORDER BY total_students DESC;

SELECT
    c.course_name,
    ROUND(AVG(e.marks), 2) AS average_marks
FROM courses c
INNER JOIN enrollments e
    ON c.course_id = e.course_id
GROUP BY c.course_id, c.course_name
ORDER BY average_marks DESC;

SELECT
    grade,
    COUNT(*) AS total_students
FROM enrollments
GROUP BY grade
ORDER BY grade;

SELECT
    status,
    COUNT(*) AS total_enrollments
FROM enrollments
GROUP BY status;

-- =========================================
-- 12. CUSTOM REPORTS
-- =========================================

-- Top 3 Students

SELECT
    CONCAT(s.first_name, ' ', s.last_name) AS student_name,
    c.course_name,
    e.marks,
    e.grade
FROM students s
INNER JOIN enrollments e
    ON s.student_id = e.student_id
INNER JOIN courses c
    ON e.course_id = c.course_id
ORDER BY e.marks DESC
LIMIT 3;

-- Active Students

SELECT
    CONCAT(s.first_name, ' ', s.last_name) AS student_name,
    c.course_name,
    e.marks,
    e.grade,
    e.status
FROM students s
INNER JOIN enrollments e
    ON s.student_id = e.student_id
INNER JOIN courses c
    ON e.course_id = c.course_id
WHERE e.status = 'Active'
ORDER BY e.marks DESC;

-- City-wise Student Report

SELECT
    city,
    COUNT(*) AS total_students
FROM students
GROUP BY city
ORDER BY total_students DESC;

-- Course Performance Report

SELECT
    c.course_name,
    COUNT(e.enrollment_id) AS total_students,
    ROUND(AVG(e.marks), 2) AS average_marks
FROM courses c
LEFT JOIN enrollments e
    ON c.course_id = e.course_id
GROUP BY c.course_id, c.course_name
ORDER BY average_marks DESC;

-- =========================================
-- 13. HAVING
-- =========================================

SELECT
    c.course_name,
    COUNT(e.enrollment_id) AS total_students
FROM courses c
LEFT JOIN enrollments e
    ON c.course_id = e.course_id
GROUP BY c.course_id, c.course_name
HAVING COUNT(e.enrollment_id) >= 2
ORDER BY total_students DESC;

-- =========================================
-- 14. CASE EXPRESSION
-- =========================================

SELECT
    student_id,
    marks,
    CASE
        WHEN marks >= 85 THEN 'Excellent'
        WHEN marks >= 70 THEN 'Good'
        ELSE 'Needs Improvement'
    END AS performance
FROM enrollments
ORDER BY marks DESC;
